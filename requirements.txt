# pywin32 是 MCP SDK 在 **Windows** 上的硬依赖：mcp.server.stdio 无条件 import
# mcp.os.win32.utilities，里面再 import pywintypes——Linux 上那段被 sys.platform
# 挡掉，Windows 上绕不过去。pywin32 的轮子按 CPython ABI 发布，解释器一换就可能
# 只剩 cp314 之类的旧 .pyd，报 "DLL load failed" / "DLL 初始化例程失败"。
# 312 同时提供 cp39–cp315 轮子，**必须显式钉住**：不写的话它只是 mcp 的传递依赖，
# 版本跟着解释器漂，Windows 开发机上换 Python 就会炸。
pywin32==312; sys_platform == "win32"

# ---- studio gRPC（V4，§22.8）----
# 技能脚本的执行通道。web_agent 是**客户端**，studio 是**服务端**。
# grpcio-tools 只在生成桩代码时用（scripts/gen_proto.py），运行期不需要——
# 但它也带着 protoc 本体，装着省事。
#
# 固定版本是**兼容性要求**，不只是“锁一下版本”：
#
# 1. 1.85.1 的 manylinux wheel 在无 AVX 的 x86_64 小主机（100.114.77.27）
#    上 import 即段错误（退出 139），会让测试在收集用例前就中止。
# 2. Python 3.15 只有 1.85.1 提供 cp315 轮子（1.76.0 最高只到 cp314）——
#    也就是说「上 3.15」和「不踩 AVX 段错误」目前互斥。
# 3. 生成桩与 runtime 必须一起升级，不能只改其中一个。
grpcio==1.76.0
grpcio-tools==1.76.0
# protobuf 是**运行期**依赖：生成的桩代码 import 的是 google.protobuf。
# 以前只靠 grpcio-tools 传递带进来——那在开发机上恰好成立，但独立构建的
# studio 镜像（只装 grpcio）会直接炸 "No module named 'google'"。
# grpcio-tools 1.76 生成的桩要求 protobuf 6.31.1；固定同一版本以免重新
# 解析依赖时越过生成代码的兼容边界。
protobuf==6.31.1
